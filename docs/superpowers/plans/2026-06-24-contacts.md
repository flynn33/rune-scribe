# Contacts & Per-Contact Ciphers — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a cipher optionally link to a `Contact` (a local snapshot), with contacts brought in via the iOS system picker / "Import all" / manually, all encrypted at rest and device-only — and the safety guarantees that Apple-contact edits/deletes never touch ciphers and unlinking/deleting a contact never deletes a cipher.

**Architecture:** A `Contact` snapshot model + a `ContactBook` collection persisted as a separate AES-GCM-encrypted file (`ContactStore`, key injectable for tests). A cipher gains an optional `contactID`. Pure `CipherLibrary` mutations (`settingContact`, `unlinkingContact`) hold the safety logic and are unit-tested; `KeyManager`/`ContactManager` delegate to them. Contacts UI + a one-time import prompt sit on top. **No rotation, no Messages** (those are other phases).

**Tech Stack:** Swift 6, SwiftUI, CryptoKit (`AES.GCM`), Contacts/ContactsUI (`CNContactStore`, `CNContactPickerViewController`), XCTest, XcodeGen. Module `RuneScrib`. Spec: `docs/superpowers/specs/2026-06-24-runescribe-phase3-contacts-design.md`.

```swift
    let createdAt: Date

         appleContactID: String? = nil, source: Source, createdAt: Date = Date()) {
        self.id = id
        self.displayName = displayName
        self.phones = phones
        self.emails = emails
        self.appleContactID = appleContactID
        self.source = source
        self.createdAt = createdAt
    }
}
```

```swift
// RuneScrib/Models/ContactBook.swift
import Foundation

/// The persisted collection of the user's contacts. Stored AES-GCM-encrypted in a file
/// (see `ContactStore`). Pure mutations return a new book and are unit-testable.
struct ContactBook: Codable, Equatable {
    var version: Int = 1
    var contacts: [Contact]

    static let empty = ContactBook(contacts: [])

    private enum CodingKeys: String, CodingKey { case version, contacts }
}

extension ContactBook {
    func contact(id: UUID) -> Contact? { contacts.first { $0.id == id } }

    func adding(_ c: Contact) -> ContactBook {
        var copy = self; copy.contacts.append(c); return copy
    }
    func updating(_ c: Contact) -> ContactBook {
        var copy = self
        if let i = copy.contacts.firstIndex(where: { $0.id == c.id }) { copy.contacts[i] = c }
        return copy
    }
    func removing(id: UUID) -> ContactBook {
        var copy = self; copy.contacts.removeAll { $0.id == id }; return copy
    }
}
```

```swift
// RuneScrib/Services/ContactStore.swift
import Foundation
import CryptoKit

/// Persists the `ContactBook` AES-GCM-encrypted in a file in app-support, using the same
/// Keychain-held symmetric key as the cipher library. The key and directory are injectable so
/// the encrypted round-trip is unit-testable without touching the Keychain or app-support.
final class ContactStore: @unchecked Sendable {
    static let shared = ContactStore()

    private let injectedKey: SymmetricKey?
    private let directory: URL?

    init(key: SymmetricKey? = nil, directory: URL? = nil) {
        self.injectedKey = key
        self.directory = directory
    }

    /// Decrypts the stored book, or `.empty` if absent/unreadable (e.g. wrong key).
    func load() -> ContactBook {
        guard let data = try? Data(contentsOf: fileURL),
              let key = try? symmetricKey(),
              let book = try? decrypt(ContactBook.self, from: data, using: key)
        else { return .empty }
        return book
    }

    /// AES-GCM-encrypts and writes the book atomically.
    func save(_ book: ContactBook) throws {
        let key = try symmetricKey()
        let data = try encrypt(book, using: key)
        try FileManager.default.createDirectory(at: fileURL.deletingLastPathComponent(),
                                                withIntermediateDirectories: true)
        try data.write(to: fileURL, options: [.atomic])
    }

    // MARK: helpers

    private var fileURL: URL {
        let base = directory ?? FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("RuneScribContacts", isDirectory: true)
        return base.appendingPathComponent("contacts.enc")
    }

    private func symmetricKey() throws -> SymmetricKey {
        if let injectedKey { return injectedKey }
        if let data = try KeychainService.shared.retrieveSymmetricKeyData() { return SymmetricKey(data: data) }
        let key = SymmetricKey(size: .bits256)
        try KeychainService.shared.storeSymmetricKeyData(key.withUnsafeBytes { Data($0) })
        return key
    }

    private func encrypt<T: Encodable>(_ value: T, using key: SymmetricKey) throws -> Data {
        let encoder = JSONEncoder(); encoder.dateEncodingStrategy = .iso8601
        let box = try AES.GCM.seal(try encoder.encode(value), using: key)
        guard let combined = box.combined else { throw KeyManagerError.encryptionFailed }
        return combined
    }
    private func decrypt<T: Decodable>(_ type: T.Type, from data: Data, using key: SymmetricKey) throws -> T {
        let box = try AES.GCM.SealedBox(combined: data)
        let plaintext = try AES.GCM.open(box, using: key)
        let decoder = JSONDecoder(); decoder.dateDecodingStrategy = .iso8601
        return try decoder.decode(T.self, from: plaintext)
    }
}
```

```swift
// RuneScrib/Services/ContactImporter.swift
import Foundation
import Contacts

/// Bridges Apple Contacts into local `Contact` snapshots. This file holds the pure mapping;
/// the system picker (iOS) and "import all" live in the UI/permission layer (added later).
enum ContactImporter {
    /// Snapshots a `CNContact` into a local `Contact` (`.imported`, keeping the identifier
    /// for reference only — never auto-resynced).
    static func snapshot(from cn: CNContact) -> Contact {
        let name = CNContactFormatter.string(from: cn, style: .fullName)?
            .trimmingCharacters(in: .whitespacesAndNewlines)
        return Contact(
            displayName: (name?.isEmpty == false ? name! : "Unnamed"),
            phones: cn.phoneNumbers.map { $0.value.stringValue },
            emails: cn.emailAddresses.map { $0.value as String },
            appleContactID: cn.identifier,
            source: .imported
        )
    }
}
```

  ```swift
  // RuneScrib/Services/ContactManager.swift
  import Foundation

  /// Owns the user's contacts (loaded/saved via `ContactStore`) and keeps cipher links consistent
  /// when a contact is deleted. Published for SwiftUI; all state on the main actor.
  @MainActor
  final class ContactManager: ObservableObject {
      static let shared = ContactManager()
      @Published private(set) var contacts: [Contact] = []
      private let store: ContactStore
      private let keyManager: KeyManager

      init(store: ContactStore = .shared, keyManager: KeyManager = .shared) {
          self.store = store; self.keyManager = keyManager
          contacts = store.load().contacts
      }

      func add(_ c: Contact) { persist(book().adding(c)) }
      func update(_ c: Contact) { persist(book().updating(c)) }

      /// Deletes a contact and unlinks (never deletes) its ciphers.
      func delete(id: UUID) {
          keyManager.unlinkContact(id)
          persist(book().removing(id: id))
      }

      func contact(id: UUID?) -> Contact? { id.flatMap { id in contacts.first { $0.id == id } } }

      private func book() -> ContactBook { ContactBook(contacts: contacts) }
      private func persist(_ updated: ContactBook) {
          contacts = updated.contacts
          try? store.save(updated)
      }
  }
  ```

    ```swift
    /// Requests Contacts access and returns snapshots of every contact with a usable name.
    static func importAll() async throws -> [Contact] {
        let store = CNContactStore()
        let granted = (try? await store.requestAccess(for: .contacts)) ?? false
        guard granted else { throw ContactImportError.accessDenied }
        let keys = [CNContactFormatter.descriptorForRequiredKeys(for: .fullName),
                    CNContactPhoneNumbersKey as CNKeyDescriptor, CNContactEmailAddressesKey as CNKeyDescriptor]
        var out: [Contact] = []
        let req = CNContactFetchRequest(keysToFetch: keys)
        try store.enumerateContacts(with: req) { cn, _ in
            let snap = snapshot(from: cn)
            if snap.displayName != "Unnamed" { out.append(snap) }
        }
        return out
    }
    ```
